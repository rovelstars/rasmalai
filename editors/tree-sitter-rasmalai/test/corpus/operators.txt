fn ops(a: Int, b: Int): Int {
  let n = a ?? b;
  let o = a | b;
  let x = a ^ b;
  let y = a & b;
  let s = a << 2;
  let t = a >> 1;
  let z = a >>> 1;
  let w = -a + !ok + ~a;
  ++a;
  b--;
  return n + o + x + y + s + t + z + w;
}

fn pick(c: Bool, x: Int, y: Int): Int {
  let t = c ? x : y;
  return t;
}

fn checks(v: Any): Bool {
  return v is Int;
}

fn conv(v: Any): Int {
  return v as Int;
}
